% PC 1: RAINIER GRIMALDI
\pgheader[GrimaldiPurple]{Rainier Grimaldi}{The Lord / The Pragmatist}{Grimaldi}
\pgquote{Monaco exists because every Grimaldi has known when to yield and when not to. Tonight I choose.}

\section*{Background \& Drive}
\begin{rulebox}{}
\textbf{Background:} Fifty-one years old, Lord of Monaco, family patriarch. He is
not in Genoa by choice --- he came because the revolt is inevitable and he wants to
be on the right side, whichever that turns out to be.

\medskip\noindent\textcolor{GrimaldiPurple}{\textbf{Drive:}} Formal recognition of
Monaco as an autonomous lordship. The revolt is the best moment to obtain it ---
from the winner, whoever that is.
\end{rulebox}

  \azione{Study}{Studio}{2}{International law, history of lordships}

  \azione{Tinker}{Ingegno}{1}{Logistics, harbour organisation}
\end{attributoblock}

  \azione{Wreck}{Forza bruta}{0}{No}

\section*{Special Ability}
\begin{grimaldibox}{The Safe Exit}
Monaco is neutral territory. Rainier may declare that any NPC or PC is
``under Monaco's protection'' --- that person is untouchable for the time
needed to leave Genoa. Works even on enemies.
\textit{Once per session.}
\end{grimaldibox}

\begin{rulebox}{}

\medskip\noindent\textcolor{GrimaldiPurple}{\textbf{Drive:}} Career. She wants to
return to Paris with intelligence good enough to secure a court position. She loves
power, and power is wherever she decides to go.

% PC 3: COSTANTINO
\pgheader[GrimaldiPurple]{Costantino Paleologos}{The Merchant / The Informant}{Grimaldi}
\pgquote{Every harbour in the Mediterranean has its price. I know all the prices.}

\section*{Background \& Drive}
\begin{rulebox}{}
\textbf{Background:} Fifty-four years old, Venetian by birth, Genoese by residence
for twenty years, Greek by blood and identity. He describes himself as a ``silk
merchant'' but trades in anything of value --- information included. The Grimaldi
are his primary client, but not his only one. His network stretches from Alexandria
to Barcelona.

\medskip\noindent\textcolor{GrimaldiPurple}{\textbf{Drive:}} Money and stability.
A free Genoa is better for trade --- the French tax everything and respect nothing.
But Costantino would survive under the French too. He always survives.
\end{rulebox}

% PC 5: SERAFINA
\pgheader[GrimaldiPurple]{Serafina Grimaldi}{The Heir / The Wild Card}{Grimaldi}
\pgquote{My father wants me safe. Safety has never freed anyone.}

\section*{Background \& Drive}
\begin{rulebox}{}
\textbf{Background:} Nineteen years old, Rainier's only child. She has studied in
Monaco with the best tutors her father could afford. She has never fought. She has
never negotiated alone. She has observed everything, from behind half-open doors,
from a corner of council tables she was not invited to attend. She knows more than
anyone gives her credit for.

\medskip\noindent\textcolor{GrimaldiPurple}{\textbf{Drive:}} To be present. Not to
be protected --- to be there, doing things. The Genoese revolt is the first real
thing she has the chance to participate in.
\end{rulebox}
